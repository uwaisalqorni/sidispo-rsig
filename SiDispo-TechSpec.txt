1. Overview Sistem
SiDispo adalah aplikasi web & mobile untuk melacak alur disposisi surat dari Direktur hingga pelaksana. Sistem ini menggantikan proses manual dengan alur digital yang transparan, real-time, dan terarsip dengan baik.

1.1 Tujuan Sistem
#
Tujuan
Manfaat
1
Direktur/pejabat tahu perkembangan disposisi real-time
Tidak perlu tanya manual ke staf
2
Dokumen terorganisir dalam folder bertingkat
Mudah ditemukan, tidak hilang
3
Notifikasi otomatis deadline & progress
Tidak ada disposisi yang terlewat
4
Timeline progress kronologis
Akuntabilitas jelas per orang
5
Laporan & rekap per unit/bulan
Data untuk evaluasi kinerja

1.2 Ruang Lingkup
Platform: Web (Desktop + Tablet) dan Mobile (PWA / Android via Capacitor)
Pengguna: Direktur, Pejabat/Kabid, Staf Pelaksana, Admin/Sekretariat
Akses: Intranet RS + VPN untuk akses dari luar, opsional cloud hosting


2. Arsitektur Sistem
2.1 Gambaran Arsitektur
Arsitektur: SPA + REST API
Frontend Vue.js 3 (SPA) berkomunikasi dengan Backend CI3 via REST API.
CI3 berperan sebagai pure API server (tidak ada view/template CI3 yang digunakan).
Authentication menggunakan JWT (JSON Web Token) yang disimpan di localStorage.
File upload disimpan di storage server, path-nya tersimpan di database.

5. Panduan Implementasi
5.1 MY_Controller.php — Base Controller CI3
Semua controller API extend dari MY_Controller. Berisi: JWT validation, CORS header, dan helper response JSON.

<?php
// application/core/MY_Controller.php
class MY_Controller extends CI_Controller {

    protected $user_id = null;
    protected $user_role = null;

    public function __construct() {
        parent::__construct();
        $this->set_cors_headers();
        $this->load->helper(&quot;response&quot;);
        $this->load->library(&quot;JWT&quot;);
    }

    protected function set_cors_headers() {
        header(&quot;Access-Control-Allow-Origin: *&quot;);
        header(&quot;Content-Type: application/json&quot;);
        header(&quot;Access-Control-Allow-Methods: GET,POST,PUT,DELETE,OPTIONS&quot;);
        header(&quot;Access-Control-Allow-Headers: Authorization,Content-Type&quot;);
        if ($_SERVER[&quot;REQUEST_METHOD&quot;] === &quot;OPTIONS&quot;) { exit(0); }
    }

    // Wajib dipanggil di setiap method yang butuh login
    protected function auth_check() {
        $token = $this->get_bearer_token();
        $payload = $this->jwt->decode($token);
        if (!$payload) { json_error(&quot;Unauthorized&quot;, 401); exit; }
        $this->user_id   = $payload->user_id;
        $this->user_role = $payload->role;
    }

    protected function get_bearer_token() {
        $header = $this->input->get_request_header(&quot;Authorization&quot;);
        return str_replace(&quot;Bearer &quot;, &quot;&quot;, $header);
    }
}

5.2 Disposisi.php — Contoh Controller
<?php
// application/controllers/Disposisi.php
class Disposisi extends MY_Controller {

    public function index_get() {
        $this->auth_check();
        $this->load->model(&quot;Disposisi_model&quot;);

        // Filter params dari query string
        $status  = $this->input->get(&quot;status&quot;);
        $folder  = $this->input->get(&quot;folder_id&quot;);

        $data = $this->Disposisi_model
                     ->get_list($this->user_id, $this->user_role, $status, $folder);

        json_success($data);
    }

    public function index_post() {
        $this->auth_check();
        // Hanya DIREKTUR dan PEJABAT yang bisa buat disposisi
        if (!in_array($this->user_role, [&quot;DIREKTUR&quot;,&quot;PEJABAT&quot;])) {
            json_error(&quot;Forbidden&quot;, 403); return;
        }
        $this->load->model(&quot;Disposisi_model&quot;);
        $this->load->library(&quot;form_validation&quot;);

        $payload = json_decode(file_get_contents(&quot;php://input&quot;), true);

        // Validasi input
        if (empty($payload[&quot;isi_disposisi&quot;]) || empty($payload[&quot;penerima&quot;])) {
            json_error(&quot;Field wajib tidak lengkap&quot;, 422); return;
        }

        $id = $this->Disposisi_model->create($payload, $this->user_id);

        // Kirim notifikasi ke penerima via FCM
        $this->load->library(&quot;FCM&quot;);
        $this->fcm->send_to_users($payload[&quot;penerima&quot;], [
            &quot;title&quot; => &quot;Disposisi Baru&quot;,
            &quot;body&quot;  => $payload[&quot;isi_disposisi&quot;],
            &quot;data&quot;  => [&quot;disposisi_id&quot; => $id]
        ]);

        json_success([&quot;id&quot; => $id], &quot;Disposisi berhasil dibuat&quot;, 201);
    }
}

5.3 Axios Instance — Frontend Vue.js
// src/api/axios.js
import axios from &apos;axios&apos;
import { useAuthStore } from &apos;@/stores/auth&apos;

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,  // dari .env
  timeout: 15000,
  headers: { &apos;Content-Type&apos;: &apos;application/json&apos; }
})

// Request interceptor: tambah JWT token otomatis
api.interceptors.request.use(config => {
  const auth = useAuthStore()
  if (auth.token) {
    config.headers.Authorization = `Bearer ${auth.token}`
  }
  return config
})

// Response interceptor: handle 401 → redirect ke login
api.interceptors.response.use(
  res => res,
  err => {
    if (err.response?.status === 401) {
      const auth = useAuthStore()
      auth.logout()
      window.location.href = &apos;/login&apos;
    }
    return Promise.reject(err)
  }
)

export default api

5.4 Pinia Store — Disposisi
// src/stores/disposisi.js
import { defineStore } from &apos;pinia&apos;
import api from &apos;@/api/axios&apos;

export const useDisposisiStore = defineStore(&apos;disposisi&apos;, {
  state: () => ({
    list: [],
    detail: null,
    loading: false,
    filter: { status: &apos;semua&apos;, folder_id: null }
  }),

  actions: {
    async fetchList() {
      this.loading = true
      try {
        const { data } = await api.get(&quot;/disposisi&quot;, {
          params: this.filter
        })
        this.list = data.data
      } finally {
        this.loading = false
      }
    },

    async fetchDetail(id) {
      const { data } = await api.get(`/disposisi/${id}`)
      this.detail = data.data
    },

    async createDisposisi(payload) {
      const { data } = await api.post(&quot;/disposisi&quot;, payload)
      await this.fetchList()
      return data
    }
  }
})

10. Keamanan & Best Practice
Aspek
Implementasi
Authentication
JWT Token (expire 8 jam). Refresh token untuk extend session.
Authorization
Role check di setiap endpoint. MY_Controller->auth_check() + role_check().
SQL Injection
Wajib gunakan CI3 Query Builder ($this->db->where(), dll). TIDAK boleh raw query string.
XSS Prevention
Semua output Vue.js menggunakan {{ }} (auto-escape). Hindari v-html kecuali terpaksa.
File Upload
Validasi mime type + ekstensi. Simpan dengan nama random (uniqid). Folder upload di luar webroot.
CORS
Batasi Access-Control-Allow-Origin ke domain FE saja di production.
Password
Hash menggunakan password_hash() PHP (bcrypt). TIDAK boleh MD5/SHA1.
HTTPS
Wajib SSL di production. Redirect HTTP ke HTTPS via .htaccess.
Rate Limiting
Batasi login attempt: maks 5x gagal dalam 10 menit, block IP sementara.
